function applyTheme() {

Q3. 메뉴·등장 애니메이션·맨 위로 버튼

⭐ 무거운 계산은 브라우저에 — 감시는 IntersectionObserver, 표시는 scroll

  • 메뉴: setState({menuOpen}) → applyMenu. 버튼 이름('메뉴')은 고정이고, 열림·닫힘은 aria-expanded(스크린리더)와 CSS 아이콘 형태(☰→✕)로 전달합니다.
  • 등장: 스크롤마다 계산하면 Reflow가 반복돼 버벅입니다. IntersectionObserver에 감시를 맡기고, 20% 지점에서 클래스를 붙인 뒤 unobserve로 감시를 끕니다 — 등장은 한 번뿐.
  • 맨 위로: 스크롤 이벤트 하나에서 300px는 버튼 표시, 60px는 내비 전환(불투명 배경+헤더 그림자)을 담당합니다. 클릭하면 scrollTo({behavior:'smooth'})로 올라갑니다.
function applyMenu() {
  nav.classList.toggle('active', STATE.menuOpen);   // 보이는 메뉴
  // 버튼 이름은 '메뉴'로 고정. 열림/닫힘은 aria-expanded(스크린리더) + CSS 아이콘 형태로 전달.
  menuButton.setAttribute('aria-expanded', String(STATE.menuOpen));
}

const watcher = new IntersectionObserver((entries) => {
  entries.forEach((entry) => {
    if (entry.isIntersecting) {
      entry.target.classList.add('visible');   // 등장
      watcher.unobserve(entry.target);         // 본 구역은 감시 종료
    }
  });
}, { threshold: 0.2 });

window.addEventListener('scroll', () => {
  topButton.classList.toggle('show', window.scrollY > 300);
  nav.classList.toggle('scrolled', window.scrollY > 60);

Q4. 저장소 목록과 로딩·에러·빈 상태

⭐ 서버가 404로 답해도 fetch는 성공

  • fetch는 서버가 404·403으로 답해도 성공으로 끝납니다. 거절은 망이 끊길 때뿐 — 그래서 res.ok를 직접 검사해 실패를 내가 만듭니다.
  • 타임아웃 8초(AbortController.abort()), 재시도는 버튼으로만 수동, 캐싱은 없습니다.
왜 실패를 '내가' 만드는가 — fetch 함정과 4가지 상태

네 갈래가 서로 다른 말을 합니다. 403 한도 초과 / 404 사용자 없음 / 그 외 HTTP 상태 코드(직접 throw) / 망 단절·타임아웃(catch에 자연 발생). "catch 하나로 다 처리"가 아니라 화면에 보여줄 문구가 갈리기 때문입니다.

타임아웃의 작동 원리: new AbortController()가 취소 조종기이고, controller.signal(취소 센서)을 fetch에 붙여둡니다. 8초 뒤 controller.abort()가 호출되면 진행 중인 요청이 취소되면서 fetch가 실패로 바뀌어 catch로 떨어집니다. fetch에는 취소 버튼이 따로 없기 때문에, "늦은 응답을 버리는" 유일한 방법이 이 신호 연결입니다.

async function loadRepos() {
  const controller = new AbortController();
  let timedOut = false;
  // ① 8초 후: 취소 표시를 남기고 요청 취소
  const timeoutId = setTimeout(() => { timedOut = true; controller.abort(); }, 8000);
  try {
    const res = await fetch(reposURL, { signal: controller.signal });
    // ② 실패를 내가 만든다 — fetch는 404·403도 성공이라 ok를 직접 검사
    if (!res.ok) {
      if (res.status === 403) throw new Error('API 호출 한도를 초과했습니다. 잠시 후 다시 시도해주세요.');
      if (res.status === 404) throw new Error('사용자를 찾을 수 없습니다.');
      throw new Error(`로드 실패 (HTTP ${res.status})`);
    }
    renderRepos(await res.json());              // ③ 성공 → 카드 렌더
  } catch (err) {
    reposStatus.textContent = timedOut          // ④ 오류 문구(타임아웃 여부로 분기)
      ? '요청 시간이 초과됐습니다. 다시 시도해주세요.' : err.message;
    reposRetry.hidden = false;                  //     + 수동 재시도 버튼
  } finally {
    window.clearTimeout(timeoutId);             // ⑤ 성공·실패 모두 타이머 정리
  }
}

라이브 데모 — 없는 파일로 404를 직접 관측

버튼이 없는 파일 no-such-file-b11.json을 실제로 요청합니다. 로그의 ①이 "fetch는 성공으로 끝난다", ②가 "실패는 내가 만든다"의 증거입니다.

대기 — 버튼을 누르면 실제 요청이 시작됩니다.

Q6. 왜 세 개의 파일인가

⭐ 변경 최소화.

  • 목적은 깔끔함이 아니라 독립된 변경입니다.
  • 분리를 유지하는 규칙: 인라인 onclick·스타일 금지, var 금지(const·let 사용), 'use strict' + DOMContentLoaded.
규칙마다 이유가 있다 — 의지력·defer·var

한 파일에 다 넣으면 "하나라만 건드린다"는 약속이 의지력이 되는데, 의지력은 반드시 무너집니다. 파일로 나누면 구조가 막아줍니다.

script 줄의 defer는 두 줄의 순서 약속입니다: HTML 해석을 막지 않고 해석이 끝난 뒤 실행. async는 중간에라도 달려들므로 DOM이 아직 없을 수 있습니다. 원문 요구사항이 defer를 지정한 이유도 여기입니다.

var 금지는 함수 스코프·호이스팅·중복 선언 문제를 상수 const와 재할당만 허용하는 let으로 원천 차단하기 위함이고, 인라인 금지 규칙은 구조 파일에 표현·동작을 되집어넣지 않게 하려는 것입니다.

<!-- 표현(CSS)은 구조(HTML)와 섞지 않고 외부 파일로 연결한다 -->
<link rel="stylesheet" href="css/style.css" />
  …
<!-- 동작(JS)도 분리한다. defer는 HTML 해석을 막지 않고,
     해석이 끝난 뒤 순서대로 실행하라는 뜻이다 -->
<script src="js/main.js" defer></script>
🔤 독립된 변경 — 하나를 고칠 때 나머지를 열 필요가 없는 상태 · 결합 지점 — 파일들이 서로를 아는 유일한 자리, 여기선 link·script 두 줄 · defer · async — 해석 끝난 뒤 실행 · 다운로드 끝나면 중간에도 실행(DOM 준비 안 됨) · DOMContentLoaded — HTML 해석이 끝나 준비됐다는 이벤트 · const · let — 재할당 불가 상수 · 재할당 가능 선언(var의 스코프·호이스팅 문제 회피) · 'use strict' — 파일 최상위 선언. 조용히 넘어가는 실수(미선언 변수 대입 등)를 즉각 오류로 바꿔 잡아준다

Q8. CSS 변수를 왜 모았나

⭐ 값이 아니라 이름을 참조 — 다크 테마는 스타일 세트 교체

  • 색·간격·모서리 값을 :root에 한 번씩 정의하고 var(--이름)으로 재사용합니다. 값을 고칠 일이 생기면 정의 한 곳만 건드립니다.
  • 다크모드는 새 스타일이 아니라 같은 이름의 다른 값입니다. [data-theme="dark"] 블록이 변수만 덮어쓰면, 변수를 참조하는 아래쪽 스타일 전부가 자동으로 따라 바뀝니다.
  • JS는 색 값을 전혀 만지지 않습니다. data-theme 속성만 바꿉니다. 표현은 CSS의 일이라는 분리가 테마 전환에서도 지켜집니다.
:root {                          /* 라이트가 기본값 */
  --bg: #ffffff;
  --text: #1a1b1e;
  --accent: #1f6feb;
}
[data-theme="dark"] {            /* 이름은 두고 값만 갈아끼움 */
  --bg: #121212;
  --text: #f1f3f5;
  --accent: #6ea8fe;
}

body { background: var(--bg); color: var(--text); }   // 참조자는 그대로
🔤 CSS 사용자 정의 속성 — 값에 --이름을 붙여 한 곳에 정의하고 재사용 · :root — 문서의 최상위 — 변수를 모으는 정의 자리 · data-* 속성 — HTML·CSS·JS가 자유 이름으로 약속을 주고받는 속성

Q9. onclick 대신 addEventListener

⭐ 각 버튼에 리스너가 아닌, 부모 리스너 - 자식 이벤트 위임

  • 구조와 동작의 분리 — HTML에 행동을 박아두면 세 파일로 나눈 의미가 무너집니다.
  • onclick은 속성 하나라 두 번째 대입이 첫 번째를 덮어씁니다. addEventListener는 여러 개를 붙일 수 있습니다. 이 페이지의 리스너는 열 곳, onclick은 0건입니다.
  • 필터 버튼은 이벤트 위임: 버튼은 data-filter 값만 싣고, 클릭은 부모 리스너가 받아 closest로 구분합니다.
// 버튼은 식별 값만 싣습니다 — 리스너를 달지 않아요
function renderFilters(repos) {
  langs.forEach((lang) => {
    const btn = document.createElement('button');
    btn.textContent = lang;
    btn.dataset.filter = lang;         // data-filter 값만
    filters.appendChild(btn);
  });
}

// 클릭은 부모(#repos-filters) 리스너 하나가 받고,
// 눌린 버튼은 closest로 고릅니다
document.querySelector('#repos-filters')
  .addEventListener('click', (event) => {
    const btn = event.target.closest('button[data-filter]');
    if (!btn) return;
    setState({ reposFilter: btn.dataset.filter });
    renderRepos(STATE.repos);
  });
data-filter와 dataset.filter — 같은 값의 두 이름

data-*는 HTML5가 요소에 임의 정보를 심어두라고 표준으로 마련한 어트리뷰트입니다. JS는 이걸 dataset이라는 전용 지도로 보여줍니다.

<button data-filter="JavaScript">   <!-- HTML에 심은 값 -->
btn.dataset.filter                    // 'JavaScript' — 읽기
btn.dataset.filter = 'CSS'            // 쓰면 어트리뷰트가 즉시 따라 바뀜

이름이 다른 이유 셋: ① data-는 표준 속성(href·class)과의 충돌을 막는 표시, ② dataset은 data-*만 모은 상자의 이름이라 접두사가 중복이라 빠짐, ③ 하이픈은 점 표기에서 빼기 연산자로 오해되니 camelCase로 다듬음 — data-user-id → dataset.userId.

🔤 이벤트 위임 — 자식 여러 개의 이벤트를 부모 리스너 하나로 받음 · closest() — 눌린 지점에서 조건에 맞는 조상을 거슬러 검색 · onclick — HTML에 직접 박는 행동 — 재대입 시 덮어써짐 · dataset — data-* 어트리뷰트를 지도처럼 보는 창. dataset.filter ≡ data-filter

Q11. async/await·try/catch의 분기 따라가기

⭐ 실패 다섯 갈래가 catch 하나에 모인다

갈래를 누르면 — 그 실패가 지나는 칸과 화면에 뜰 문구가 바로 표시됩니다
  • try 안의 순서: 로딩 → fetch(url, {signal}) → res.ok → json() → 렌더. 단계마다 실패 지점이 다릅니다.
  • 실패 넷에 문구 넷: 403 한도 / 404 없음 / 그 외 HTTP / 망 단절·타임아웃. "catch 하나로 다 처리"가 아닙니다.
타임아웃 3단계 — 표시를 왜 먼저 적고, 정리는 왜 무조건인가

setTimeout 콜백에서 ① timedOut = true를 먼저 적고 ② controller.abort()로 fetch를 취소합니다. ③ 취소된 fetch는 AbortError로 catch에 떨어지는데, 망 단절과 타임아웃이 같은 경로로 오므로 ①의 표시로 문구를 나눕니다.

응답이 먼저 와도 예약된 취소는 남습니다. 그래서 finally에서 clearTimeout — 성공해도, 실패해도 남는 게 없도록.

const controller = new AbortController();
let timedOut = false;
const timeoutId = setTimeout(() => { timedOut = true; controller.abort(); }, 8000);
// fetch는 { signal: controller.signal }과 함께 살아 있다
// 실패 지점과 문구 분기는 Q4 코드 참조
🔤 async · await — 시간 걸리는 일을 윗줄→아랫줄 읽듯이 쓰는 문법 · try · catch · finally — 시도 → 오류 처리 → 무조건 마무리 · throw — catch로 넘길 오류를 직접 만들어 던짐

// 그래서 심기 전에 안전 문자로 바꿉니다 (XSS 방지).
function escapeHTML(value) {
  return String(value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

② escapeHTML 통과 — 우리 코드

Q13. Flex와 Grid는 어디에, 왜

⭐ 축이 하나면 Flex, 둘이면 Grid

  • 좌우 정렬만 필요한 한 줄들(헤더·내비·필터)은 1차원 — Flex로 충분합니다.
  • 저장소 카드는 행과 열이 같이 굴러가는 2차원 — Grid. 넓어지면 auto-fit이 열을 늘리고, 1200px 이상은 3열로 명시 상한을 둡니다.
  • auto-fit를 쓴 이유: 카드가 적어도 빈 트랙을 접어 채웁니다. auto-fill이었다면 빈칸을 남겨 카드가 한쪽으로 쏠립니다.
/* 한 축 → Flex */
header { display: flex; align-items: center; }
/* 링크 4개를 폭 전체에 균등 배치 */
nav    { justify-content: space-between; flex: 1; }
footer { display: flex; justify-content: space-between; }

/* 두 축 → Grid */
#repos-list { display: grid; grid-template-columns: 1fr; }  /* 기본 1열 */
@media (min-width: 768px) {
  #repos-list { grid-template-columns: repeat(auto-fit, minmax(260px, 1fr)); }
}
@media (min-width: 1200px) {
  #repos-list { grid-template-columns: repeat(3, 1fr); }
}
🔤 Flex — 한 줄(1차원) 정렬 도구 · Grid — 행과 열(2차원) 격자 도구 · space-between — 요소들을 폭 전체에 균등 배치 · auto-fill — auto-fit과 달리 격자 빈칸을 그대로 둠

Q14. STATE 객체가 왜 필요한가

⭐ 화면 결정 값의 저장소 · 바꿀 수 있는 유일한 통로

  • 화면 결정 값 넷(theme·menuOpen·repos·reposFilter)을 한 객체에, 변경 통로 setState 하나에, 그리는 함수는 읽기만.
  • 변수로 흩으면 핸들러가 늘수록 원인을 역추적하기 어려워집니다. 변경 지점이 한 곳이라 디버그 로그도 한 곳에 붙었습니다.
  • 정직한 예외 두 개(도식 아래 점선): 폼의 valid는 제출 단위 1회용, 스크롤 표시는 지나가는 신호 — 모을 값과 아닐 값을 구분한 증거입니다.
🔤 STATE · setState — 화면 결정 값의 저장소 · 바꾸는 유일한 통로 · 단일 진실 원천 — 같은 값의 원본을 한 곳에 두는 설계 · 지역 변수 — 함수 안에서만 사는 1회용 값

Q15. 모바일 퍼스트로 쓴 이유

⭐ 제약이 제일 빡빡한 대상을 기본에 두기 — 나머지는 다 그 위의 덧붙이기

  • 기본이 곧 좁은 화면(내비 접힘·격자 1열), 넓어지는 분기 min-width 셋, 되돌리는 max-width는 0개입니다.
  • 이 순서를 고른 진짜 이유: 모바일이 이 페이지에서 가장 어려운 구현 대상입니다. 공간이 좁아 레이아웃이 강제 선별되고, 터치 손가락을 위해 버튼 높이 44px(min-height: 2.75rem)를 확보해야 합니다.
  • 가치 기준의 선택이기도 합니다: 저사양·터치가 먼저 보장됩니다. Q1이 '동작'을 물었다면 이 장은 그 순서를 고른 '이유'입니다.
제일 빡빡한 제약을 먼저 풀면 무엇이 달라지나 — 검증 방향의 차이

제약이 가장 심한 화면을 먼저 풀면 데스크톱은 새 설계가 아니라 덧셈입니다. 실제로 768px 분기가 하는 일은 접어둔 내비를 다시 펼치고 열 개수를 늘리는 것뿐입니다.

순서를 뒤집으면 좁아질 때마다 이미 얹어둔 스타일을 max-width 분기로 하나씩 벗기면서 그 폭에서 다시 검증해야 합니다. 기본이 좁으면 분기가 덧셈 한쪽뿐이라, 이미 검증한 좁은 화면이 넓은 화면 분기 때문에 무너질 일이 없습니다. '계산 하나가 사라진다'의 정체도 이 검증 방향입니다.

/* 기본 = 모바일 */
nav         { display: none; }        /* 햄버거로 접음 */
#tech ul    { grid-template-columns: 1fr; }
#repos-list { grid-template-columns: 1fr; }

/* 넓어질 때만 덧입힘 — 되돌리는 분기 없음 */
@media (min-width: 768px)  { nav { display: flex; } ... }
@media (min-width: 1024px) { main { max-width: 1024px; } ... }
@media (min-width: 1200px) { main { max-width: 1200px; } ... }
🔤 min-width — 화면이 넓어질 때 도는 분기 조건 · max-width — 좁아질 때 되돌리는 분기 조건 — 이 페이지엔 없음 · 모바일 퍼스트 — 저사양·터치 환경이 먼저 보장되는 순서 · 프로그레시브 증강 — 가장 빡빡한 제약을 기본에서 풀고 쉬운 쪽으로 얹어가는 방식. 모바일 퍼스트의 정식 명칭
